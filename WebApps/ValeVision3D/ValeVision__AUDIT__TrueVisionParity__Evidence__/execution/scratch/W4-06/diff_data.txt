>     // ------------------------------------------------------------
>     // K1 DR-11, R6 F.8 C21. A statement's file name carries a phase segment
>     // only once Vale's own stages are supplied - the Drawing Register's phase
>     // list, the same one a drawing's document code takes its phase from.
>     // Until then the "_T{tranche}" part of FilePattern is dropped, so a new
>     // statement is 3047_S01__Doous__<Title>__.md and never takes NA's T01.
>     // A FilePattern with no tranche passes through unchanged.
>     // ------------------------------------------------------------
>     function Na__LeStmt__NamingSetup() {
>         const setup  = Na__LeCfg__GetStatementSetup();
>         const phases = Na__LeCfg__GetDrawingRegisterSetup().phases;
>         if (Array.isArray(phases) && phases.length) return setup;
>         return Object.assign({}, setup, { filePattern : String(setup.filePattern).replace(/_[A-Za-z]*\{tranche\}/g, '') });
>     }

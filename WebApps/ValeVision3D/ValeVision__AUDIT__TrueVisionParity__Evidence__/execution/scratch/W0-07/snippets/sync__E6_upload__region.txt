def na_upload_project_json_to_r2(s3_client, bucket: str, project_json_path: Path, r2_prefix: str, report: SyncReport):
    """Upload project.json to R2 under the given prefix without erasing what the
    ValeVision editor saved there.

    The editor writes its keys to R2 first (dev menus, presentation scenes,
    cross sections, Layout Editor sheets), so the local file can be behind R2 -
    a failed mirror, a second machine. The shared lib therefore takes every key
    on the ProjectData__EditorOwnedKeys list (ValeVision3D Na__AppConfig__Main.json)
    that R2 holds from R2's copy, this sync's IMG-slot thumbnail re-point is
    applied to that copy of PresentationMode__SavedCameraScenes, and the merged
    document is written over the local file (so it receives R2's editor data)
    and uploaded. Nothing is uploaded when the key list or R2's copy cannot be
    read: the step fails and says why.

    CacheControl is set to 'no-cache, max-age=0' (matching the build manifest /
    masterConfig mirror) so a re-synced project.json is revalidated on every
    fetch instead of potentially being served stale from the browser's HTTP
    cache or Cloudflare's edge cache — the editor Worker save path sets the
    same header; this keeps the SketchUp Cloud Sync path consistent with it.
    """
    label = 'Upload project.json to R2'
    if not project_json_path.exists():
        report.add_step(label, False, f'project.json not found: {project_json_path}')
        return

    key = f"{r2_prefix}/{PROJECT_JSON_FILENAME}"
    try:
        import AutomationUtil__R2Common__Lib__ as r2lib       # @delegate: ./AutomationUtil__R2Common__Lib__.py
    except Exception as exc:
        report.add_step(label, False, f'project.json NOT uploaded - shared R2 lib unavailable: {exc}')
        return

    wcp_project_dir = project_json_path.parent
    result = r2lib.na_upload_project_json_preserving_editor_keys(
        s3_client, bucket, project_json_path, key,
        transform=lambda document: na_repoint_merged_scene_thumbnails(document, wcp_project_dir)
    )
    report.add_step(label, result['ok'], result['message'],
                    keptFromR2=result['kept'], r2State=result['r2State'], localUpdated=result['localUpdated'])
    if result['uploaded']:
        report.uploaded += 1
